create database may_flight;
-- drop table airport;
create table airport(
    code int unique not null PRIMARY KEY,
    country varchar(128) not null,
    city varchar(128) not null
);
-- drop table aircraft;
create table aircraft(
    id serial PRIMARY KEY,
    model varchar(128) not null
);
-- drop table seat;
create table seat(
    aircraft_id int not null references aircraft PRIMARY KEY,
    seat_no int not null
);
-- drop table flight;
create table flight(
    id serial PRIMARY KEY,
    flight_no int not null,
    departure_date date,
    departure_airport_code int not null references airport(code),
    arrival_date date,
    arrival_airport_code int not null references airport(code),
    aircraft_id int not null references aircraft,
    status varchar(128) not null
);
-- drop table ticket;
create table ticket(
    id serial PRIMARY KEY,
    passport_no varchar(64) unique not null,
    passenger_name varchar(64) not null,
    flight_id int not null references flight,
    seat_no char(4) unique not null,
    cost numeric(8,2)
);
